8. Full Roc Code Sample

Realistic Roc code with mixed syntax

app [main!] { pf: platform "./platform/main.roc" }

import pf.Stdout

# Calculate the sum of a list using a for loop
for_loop = |num_list| {
	var $sum = 0

	for num in num_list {
		$sum = $sum + num
	}

	$sum
}

simple_match : [Red, Green, Blue] -> Str
simple_match = |color| {
	match color {
		Red => "The color is red."
		Green => "The color is green."
		Blue => "The color is blue."
	}
}

number_operators : I64, I64 -> I64
number_operators = |a, b| {
	sum = a + b
	diff = a - b
	prod = a * b
	eq = a == b
	neq = a != b
	sum
}

main! = || {
	Stdout.line!("Hello, world!")

	sum = for_loop([1, 2, 3, 4, 5])
	expect sum == 15

	# Print the result
	Stdout.line!(Str.inspect(sum))

	if sum > 10
		Stdout.line!("Big number")
	else
		Stdout.line!("Small number")
}
